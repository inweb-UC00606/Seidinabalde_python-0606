equipas = []

def registar_equipa():
    print("\n--- Registar Equipa ---")
    numero = input("Número da equipa: ")
    piloto = input("Nome do piloto: ")
    copiloto = input("Nome do co-piloto: ")

    equipas.append({
        "numero": numero,
        "piloto": piloto,
        "copiloto": copiloto,
        "tempo": None
    })
    print("Equipa registada com sucesso!\n")


def registar_tempo():
    print("\n--- Registar Tempo da Equipa ---")
    numero = input("Número da equipa: ")

    for e in equipas:
        if e["numero"] == numero:
            tempo = float(input("Tempo total da prova (em horas): "))
            e["tempo"] = tempo
            print("Tempo registado com sucesso!\n")
            return

    print("Equipa não existe!\n")


def mostrar_vencedor():
    print("\n--- Vencedores ---")

    # Filtrar apenas equipas com tempo registado
    equipas_com_tempo = [e for e in equipas if e["tempo"] is not None]

    if not equipas_com_tempo:
        print("Ainda não existem tempos registados.\n")
        return

    vencedor = min(equipas_com_tempo, key=lambda x: x["tempo"])

    print(f"Vencedores:")
    print(f"Número: {vencedor['numero']}")
    print(f"Piloto: {vencedor['piloto']}")
    print(f"Co-piloto: {vencedor['copiloto']}")
    print(f"Tempo: {vencedor['tempo']} horas\n")


def listar_equipas():
    print("\n--- Lista de Equipas ---")
    if not equipas:
        print("Nenhuma equipa registada.\n")
        return

    for e in equipas:
        print(f"Numero: {e['numero']} | Piloto: {e['piloto']} | Co-piloto: {e['copiloto']}")
    print()


def listar_lentos():
    print("\n--- Equipas com tempo superior a 4 horas ---")
    lentos = [e for e in equipas if e["tempo"] is not None and e["tempo"] > 4]

    if not lentos:
        print("Nenhuma equipa com tempo superior a 4 horas.\n")
        return

    for e in lentos:
        print(f"Numero: {e['numero']} | Piloto: {e['piloto']} | Co-piloto: {e['copiloto']} | Tempo: {e['tempo']}h")
    print()


# ------------------ MENU PRINCIPAL ------------------

while True:
    print("===== MENU RALLY =====")
    print("1 - Registar dupla de pilotos")
    print("2 - Registar tempo da equipa")
    print("3 - Mostrar vencedores")
    print("4 - Listar equipas")
    print("5 - Listar equipas lentas (>4h)")
    print("6 - Sair")

    opcao = input("Escolha uma opção: ")

    if opcao == "1":
        registar_equipa()
    elif opcao == "2":
        registar_tempo()
    elif opcao == "3":
        mostrar_vencedor()
    elif opcao == "4":
        listar_equipas()
    elif opcao == "5":
        listar_lentos()
    elif opcao == "6":
        print("A sair do programa... Até já!")
        break
    else:
        print("Opção inválida!\n")
